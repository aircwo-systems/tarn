package engine

import (
	"context"
	"fmt"
	"net"
	"testing"
	"time"
)

func TestReviewAlreadyReadyPortWaitsForFirstTick(t *testing.T) {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	defer listener.Close()
	go func() {
		for {
			conn, err := listener.Accept()
			if err != nil {
				return
			}
			conn.Close()
		}
	}()
	port := fmt.Sprint(listener.Addr().(*net.TCPAddr).Port)
	inv := NewInvoker()
	for range 3 {
		ctx, cancel := context.WithTimeout(context.Background(), time.Second)
		start := time.Now()
		err := inv.WaitForReady(ctx, port)
		elapsed := time.Since(start)
		cancel()
		if err != nil {
			t.Fatal(err)
		}
		t.Logf("already-listening port readiness took %s", elapsed)
		if elapsed > 50*time.Millisecond {
			t.Errorf("readiness waited for first polling tick")
		}
	}
}
