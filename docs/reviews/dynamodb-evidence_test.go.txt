func TestReviewFilteredLimitAllowsEmptyPage(t *testing.T) {
	for _, op := range []string{"scan", "query"} {
		t.Run(op, func(t *testing.T) {
			s := reviewStore(t)
			reviewPut(t, s, "01", false)
			reviewPut(t, s, "02", true)
			values := map[string]any{":p": map[string]any{"S": "p"}, ":yes": map[string]any{"BOOL": true}}
			var items []map[string]any
			var last map[string]any
			if op == "scan" {
				out, err := s.Scan("orders", "", "keep = :yes", "", nil, values, 1, nil)
				if err != nil {
					t.Fatal(err)
				}
				items, last = out.Items, out.LastEvaluatedKey
			} else {
				out, err := s.Query("orders", "", "pk = :p", "keep = :yes", "", nil, values, 1, nil, true)
				if err != nil {
					t.Fatal(err)
				}
				items, last = out.Items, out.LastEvaluatedKey
			}
			if len(items) != 0 {
				t.Errorf("page returned %d items, want zero after examining only filtered-out first item", len(items))
			}
			want := map[string]any{"pk": map[string]any{"S": "p"}, "sk": map[string]any{"S": "01"}}
			if !reflect.DeepEqual(last, want) {
				t.Errorf("last key=%v, want evaluated first-item key %v", last, want)
			}
		})
	}
}

func TestReviewIndexProjectionIncludesIndexKeys(t *testing.T) {
	for _, projection := range []string{"KEYS_ONLY", "INCLUDE"} {
		for _, index := range []string{"gsi-type", "lsi-status"} {
			t.Run(index+"/"+projection, func(t *testing.T) {
				cfg := config.Default()
				cfg.PersistenceEnabled = false
				s := NewStore(cfg)
				table := testTableDefinition()
				table.StreamSpecification = nil
				table.GlobalSecondaryIndexes[0].Projection = types.DynamoDBProjection{ProjectionType: projection, NonKeyAttributes: []string{"keep"}}
				table.LocalSecondaryIndexes[0].Projection = types.DynamoDBProjection{ProjectionType: projection, NonKeyAttributes: []string{"keep"}}
				if _, err := s.CreateTable(table); err != nil {
					t.Fatal(err)
				}
				reviewPut(t, s, "01", true)
				expr := "gsi_pk = :key"
				value := "g"
				keys := []string{"pk", "sk", "gsi_pk", "gsi_sk"}
				if index == "lsi-status" {
					expr = "pk = :key"
					value = "p"
					keys = []string{"pk", "sk", "lsi_sk"}
				}
				out, err := s.Query("orders", index, expr, "", "", nil, map[string]any{":key": map[string]any{"S": value}}, 0, nil, true)
				if err != nil || len(out.Items) != 1 {
					t.Fatalf("query: %v %v", out, err)
				}
				for _, key := range keys {
					if _, ok := out.Items[0][key]; !ok {
						t.Errorf("index projection omitted key %q: %v", key, out.Items[0])
					}
				}
			})
		}
	}
}

func TestReviewNumericKeysUseNumericIdentity(t *testing.T) {
	cfg := config.Default()
	cfg.PersistenceEnabled = false
	s := NewStore(cfg)
	table := &types.DynamoDBTable{TableName: "numbers", AttributeDefinitions: []types.DynamoDBAttributeDefinition{{AttributeName: "id", AttributeType: "N"}}, KeySchema: []types.DynamoDBKeySchemaElement{{AttributeName: "id", KeyType: "HASH"}}}
	if _, err := s.CreateTable(table); err != nil {
		t.Fatal(err)
	}
	item := map[string]any{"id": map[string]any{"N": "1"}}
	if _, err := s.PutItem("numbers", item, "", nil, nil, "NONE"); err != nil {
		t.Fatal(err)
	}
	out, err := s.GetItem("numbers", map[string]any{"id": map[string]any{"N": "1.0"}}, "", nil)
	if err != nil {
		t.Fatal(err)
	}
	if out.Item == nil {
		t.Error("numerically equal key 1.0 could not read stored key 1")
	}
	if _, err := s.PutItem("numbers", map[string]any{"id": map[string]any{"N": "1.0"}}, "", nil, nil, "NONE"); err != nil {
		t.Fatal(err)
	}
	described, err := s.DescribeTable("numbers")
	if err != nil {
		t.Fatal(err)
	}
	if described.ItemCount != 1 {
		t.Errorf("numerically equal keys stored %d items, want 1", described.ItemCount)
	}
}

func TestReviewDeletedCursorDoesNotRestart(t *testing.T) {
	s := reviewStore(t)
	for _, sk := range []string{"01", "02", "03"} {
		reviewPut(t, s, sk, true)
	}
	values := map[string]any{":p": map[string]any{"S": "p"}}
	first, err := s.Query("orders", "", "pk = :p", "", "", nil, values, 2, nil, true)
	if err != nil || len(first.Items) != 2 {
		t.Fatalf("first: %v %v", first, err)
	}
	if _, err := s.DeleteItem("orders", first.LastEvaluatedKey, "", nil, nil, "NONE"); err != nil {
		t.Fatal(err)
	}
	next, err := s.Query("orders", "", "pk = :p", "", "", nil, values, 2, first.LastEvaluatedKey, true)
	if err != nil {
		t.Fatal(err)
	}
	if len(next.Items) != 1 || !reflect.DeepEqual(next.Items[0]["sk"], map[string]any{"S": "03"}) {
		t.Fatalf("deleted cursor restarted query: %v", next.Items)
	}
}

// Test the actual HTTP adapter with the original module, not the scratch Store.
func TestReviewWireScannedCountBeforeFilter(t *testing.T) {
	for _, action := range []string{"Scan", "Query"} {
		t.Run(action, func(t *testing.T) {
			cfg := config.Default()
			cfg.PersistenceEnabled = false
			svc := actual.NewService(cfg)
			t.Cleanup(svc.Close)
			table := testTableDefinition()
			table.StreamSpecification = nil
			if _, err := svc.CreateTable(table); err != nil {
				t.Fatal(err)
			}
			for i := range 2 {
				item := map[string]any{"pk": map[string]any{"S": "p"}, "sk": map[string]any{"S": fmt.Sprint(i)}, "keep": map[string]any{"BOOL": i == 1}}
				if _, err := svc.PutItem("orders", item, "", nil, nil, "NONE"); err != nil {
					t.Fatal(err)
				}
			}
			req := map[string]any{"TableName": "orders", "FilterExpression": "keep = :yes", "ExpressionAttributeValues": map[string]any{":yes": map[string]any{"BOOL": true}}}
			if action == "Query" {
				req["KeyConditionExpression"] = "pk = :p"
				req["ExpressionAttributeValues"].(map[string]any)[":p"] = map[string]any{"S": "p"}
			}
			data, err := json.Marshal(req)
			if err != nil {
				t.Fatal(err)
			}
			r := httptest.NewRequest(http.MethodPost, "http://localhost/", strings.NewReader(string(data)))
			r.Header.Set("X-Amz-Target", "DynamoDB_20120810."+action)
			rec := httptest.NewRecorder()
			api.NewHandler(svc).Dispatch(rec, r)
			var out struct {
				Count        int
				ScannedCount int
			}
			if err := json.Unmarshal(rec.Body.Bytes(), &out); err != nil {
				t.Fatal(err)
			}
			if rec.Code != 200 || out.Count != 1 || out.ScannedCount != 2 {
				t.Fatalf("response=%d %s, want Count=1 ScannedCount=2", rec.Code, rec.Body.String())
			}
		})
	}
}

func TestReviewCloneParityAndIsolation(t *testing.T) {
	cases := []map[string]any{
		nil, {}, {"S": map[string]any{"S": "value"}, "N": map[string]any{"N": "1.00"}, "M": map[string]any{"M": map[string]any{"x": map[string]any{"L": []any{nil, true, float64(1)}}}}},
		{"map": map[string]any(nil), "list": []any(nil), "empty": []any{}},
		{"typed": []string{"a", "b"}, "bytes": []byte{1, 2}, "integer": int64(3), "number": json.Number("4")},
		{"badUTF8": string([]byte{0xff})}, {string([]byte{0xff}): "invalid key"}, {"badNumber": math.NaN()}, {"unsupported": func() {}},
	}
	for i, item := range cases {
		t.Run(fmt.Sprint(i), func(t *testing.T) {
			var expected map[string]any
			if item != nil {
				roundTripClone(item, &expected)
			}
			got := cloneItem(item)
			if !reflect.DeepEqual(got, expected) {
				t.Fatalf("clone changed normalization: got=%#v want=%#v", got, expected)
			}
		})
	}
	source := map[string]any{"nested": map[string]any{"L": []any{map[string]any{"S": "original"}}}}
	out := cloneItem(source)
	out["nested"].(map[string]any)["L"].([]any)[0].(map[string]any)["S"] = "changed"
	if source["nested"].(map[string]any)["L"].([]any)[0].(map[string]any)["S"] != "original" {
		t.Fatal("clone shares nested mutable data")
	}
}

func TestReviewCachedSortPreservesCurrentComparator(t *testing.T) {
	s := reviewStore(t)
	state := s.tables["orders"]
	for i, value := range []any{map[string]any{"S": "A"}, map[string]any{"S": string(rune(10))}, map[string]any{"S": "z"}, map[string]any{"S": "é"}, map[string]any{"N": "2"}, map[string]any{"N": "10"}, map[string]any{"N": "2.0"}, map[string]any{"B": "AQI="}} {
		item := testItem(fmt.Sprint(i), fmt.Sprintf("%04d", i), "OPEN", i, "1")
		item["gsi_pk"] = map[string]any{"S": "g"}
		item["gsi_sk"] = value
		key, err := buildKeyMap(item, state.Table.KeySchema)
		if err != nil {
			t.Fatal(err)
		}
		id, err := serializeKey(key, state.Table.KeySchema)
		if err != nil {
			t.Fatal(err)
		}
		state.Items[id] = itemState{Item: item}
	}
	for _, index := range []string{"", "lsi-status", "gsi-type"} {
		for _, reverse := range []bool{false, true} {
			t.Run(fmt.Sprintf("%s/%t", index, reverse), func(t *testing.T) {
				schema, _, err := lookupIndex(state.Table, index)
				if err != nil {
					t.Fatal(err)
				}
				var expected []itemView
				for _, stored := range state.Items {
					key, err := buildKeyMap(stored.Item, unionKeySchema(state.Table.KeySchema, schema))
					if err != nil {
						t.Fatal(err)
					}
					expected = append(expected, itemView{item: cloneItem(stored.Item), key: key})
				}
				sort.Slice(expected, func(i, j int) bool {
					cmp := compareForSchema(expected[i].item, expected[j].item, schema)
					if cmp == 0 {
						cmp = compareForSchema(expected[i].item, expected[j].item, state.Table.KeySchema)
					}
					if reverse {
						return cmp > 0
					}
					return cmp < 0
				})
				got, err := s.collectSortedItems(state, index, reverse)
				if err != nil {
					t.Fatal(err)
				}
				if len(got) != len(expected) {
					t.Fatalf("size %d want %d", len(got), len(expected))
				}
				for i := range got {
					if !reflect.DeepEqual(got[i].item, expected[i].item) || !reflect.DeepEqual(got[i].key, expected[i].key) {
						t.Fatalf("changed ordering at %d: got %v want %v", i, got[i], expected[i])
					}
				}
			})
		}
	}
}
