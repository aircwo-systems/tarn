func reviewFunctionCall(t *testing.T, method, path string, body any) (int, http.Header, []byte) {
	t.Helper()
	var payload []byte
	if body != nil {
		var err error
		payload, err = json.Marshal(body)
		if err != nil {
			t.Fatal(err)
		}
	}
	req, err := http.NewRequest(method, endpoint+path, bytes.NewReader(payload))
	if err != nil {
		t.Fatal(err)
	}
	req.Header.Set("Content-Type", "application/json")
	client := &http.Client{Timeout: 30 * time.Second}
	resp, err := client.Do(req)
	if err != nil {
		t.Fatal(err)
	}
	defer resp.Body.Close()
	data, err := io.ReadAll(resp.Body)
	if err != nil {
		t.Fatal(err)
	}
	return resp.StatusCode, resp.Header, data
}

func reviewFunctionCleanup(t *testing.T, path string) {
	t.Helper()
	t.Cleanup(func() {
		status, _, body := reviewFunctionCall(t, http.MethodDelete, path, nil)
		if status != http.StatusNoContent {
			t.Errorf("cleanup %s: %d %s", path, status, body)
		}
	})
}

func TestReviewFunctionTwoLayersExecute(t *testing.T) {
	name := fmt.Sprintf("review-two-layers-%d", time.Now().UnixNano())
	var layers []string
	for i := range 2 {
		layer := fmt.Sprintf("%s-layer%d", name, i)
		code := createZip(t, map[string]string{fmt.Sprintf("nodejs/node_modules/helper%d/index.js", i): fmt.Sprintf("module.exports.value = %d;", i)})
		status, _, body := reviewFunctionCall(t, http.MethodPost, "/2015-03-31/layers/"+layer+"/versions", map[string]any{"Content": map[string]string{"ZipFile": base64.StdEncoding.EncodeToString(code)}})
		if status != http.StatusCreated {
			t.Fatalf("publish: %d %s", status, body)
		}
		var v struct {
			LayerVersionArn string
			Version         int
		}
		if err := json.Unmarshal(body, &v); err != nil {
			t.Fatal(err)
		}
		layers = append(layers, v.LayerVersionArn)
		reviewFunctionCleanup(t, fmt.Sprintf("/2015-03-31/layers/%s/versions/%d", layer, v.Version))
	}
	code := createZip(t, map[string]string{"index.js": `exports.handler = async () => ({sum: require("helper0").value + require("helper1").value});`})
	path := "/2015-03-31/functions/" + name
	status, _, body := reviewFunctionCall(t, http.MethodPost, "/2015-03-31/functions", map[string]any{"FunctionName": name, "Runtime": "nodejs20.x", "Handler": "index.handler", "Timeout": 15, "MemorySize": 128, "Layers": layers, "Code": map[string]string{"ZipFile": base64.StdEncoding.EncodeToString(code)}})
	if status != http.StatusCreated {
		t.Fatalf("create: %d %s", status, body)
	}
	reviewFunctionCleanup(t, path)
	status, headers, body := reviewFunctionCall(t, http.MethodPost, path+"/invocations", map[string]any{})
	if status != http.StatusOK || headers.Get("X-Amz-Function-Error") != "" {
		t.Fatalf("two-layer invoke: %d functionError=%s body=%s", status, headers.Get("X-Amz-Function-Error"), body)
	}
	var result struct{ Sum int }
	if err := json.Unmarshal(body, &result); err != nil {
		t.Fatal(err)
	}
	if result.Sum != 1 {
		t.Fatalf("two-layer result: %s", body)
	}
}

func TestReviewFunctionTimeoutPoisonsNextInvoke(t *testing.T) {
	name := fmt.Sprintf("review-timeout-%d", time.Now().UnixNano())
	code := createZip(t, map[string]string{"index.js": `exports.handler = async (event) => { console.log("begin", event.id); await new Promise(resolve => setTimeout(resolve,event.delay)); console.log("end",event.id); return {id:event.id}; };`})
	path := "/2015-03-31/functions/" + name
	status, _, body := reviewFunctionCall(t, http.MethodPost, "/2015-03-31/functions", map[string]any{"FunctionName": name, "Runtime": "nodejs20.x", "Handler": "index.handler", "Timeout": 1, "MemorySize": 128, "Code": map[string]string{"ZipFile": base64.StdEncoding.EncodeToString(code)}})
	if status != http.StatusCreated {
		t.Fatalf("create: %d %s", status, body)
	}
	reviewFunctionCleanup(t, path)
	status, headers, body := reviewFunctionCall(t, http.MethodPost, path+"/invocations", map[string]any{"id": "prime", "delay": 0})
	if status != http.StatusOK || headers.Get("X-Amz-Function-Error") != "" {
		t.Fatalf("prime: %d functionError=%s body=%s", status, headers.Get("X-Amz-Function-Error"), body)
	}
	status, headers, body = reviewFunctionCall(t, http.MethodPost, path+"/invocations", map[string]any{"id": "slow", "delay": 3000})
	t.Logf("slow: %d functionError=%s body=%s", status, headers.Get("X-Amz-Function-Error"), body)
	if headers.Get("X-Amz-Function-Error") == "" {
		t.Fatalf("expected slow invoke timeout: %d %s", status, body)
	}
	start := time.Now()
	status, headers, body = reviewFunctionCall(t, http.MethodPost, path+"/invocations", map[string]any{"id": "fast", "delay": 0})
	t.Logf("fast after timeout: duration=%s status=%d functionError=%s body=%s", time.Since(start), status, headers.Get("X-Amz-Function-Error"), body)
	if status != http.StatusOK || headers.Get("X-Amz-Function-Error") != "" {
		t.Fatalf("zero-delay invocation failed after previous timeout: %d functionError=%s body=%s", status, headers.Get("X-Amz-Function-Error"), body)
	}
}
